Build and check

import { z } from "zod";

const User = z.object({ name: z.string().min(1), email: z.email() });
type User = z.infer<typeof User>;

User.parse(data);              // value, or throws ZodError
const r = User.safeParse(data); // r.success, r.data, r.error (lab)

Unions and choices

const Shape = z.discriminatedUnion("kind", [
  z.object({ kind: z.literal("circle"), radius: z.number() }),
  z.object({ kind: z.literal("square"), side: z.number() }),
]);                                  // picks the branch by "kind" (lab)

const Status = z.enum(["open", "closed"]);   // fixed strings (lab)
const Id = z.union([z.string(), z.number()]); // first match wins (lab)

Transform and refine

const Trimmed = z.string().transform((s) => s.trim());   // output type changes (lab)
type In  = z.input<typeof Trimmed>;   // string
type Out = z.output<typeof Trimmed>;  // string here; Date for a Date transform

const Range = z.object({ min: z.number(), max: z.number() })
  .refine((v) => v.min <= v.max, { error: "min must not exceed max", path: ["max"] }); // (lab)

Coercion

z.coerce.number().int().positive();   // "3" becomes 3 (lab)
z.coerce.number().nonnegative();      // "" becomes 0, and passes (lab: the trap)

z.preprocess((v) => (v === "" ? undefined : v),
  z.coerce.number().int().positive()); // "" is now rejected as missing (lab)
