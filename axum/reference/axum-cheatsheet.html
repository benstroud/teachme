Skeleton

use axum::{routing::get, Router};

#[tokio::main]
async fn main() {
    let app = Router::new().route("/", get(|| async { "Hello, World!" }));
    let listener = tokio::net::TcpListener::bind("127.0.0.1:3000").await.unwrap();
    axum::serve(listener, app).await.unwrap();
}

Routing

PatternMeaning
.route("/users", get(a).post(b))Two methods on one path
.route("/users/{id}", ...)Path parameter (0.8 syntax)
.route("/files/{*rest}", ...)Wildcard over several segments (docs)
.nest("/api", api())Mount a router under a prefix (prefix stripped inside)
.merge(other)Combine routers at one level; panics if both have a fallback (docs)
.fallback(handler)Runs only when no path matches
verified: /:id panics with Path segments must not start with :. For capture groups, use {capture}.

Order rule (verified): the body-consuming extractor (String, Json) is the last argument. Otherwise: E0277, "the trait bound ... Handler<_, _> is not satisfied".

Responses

async fn a() -> Json<User> { ... }                          // 200, application/json
async fn b() -> (StatusCode, Json<User>) { ... }              // status set by the tuple
async fn c() -> impl IntoResponse {
    ([(header::CONTENT_TYPE, "text/plain")], "hello")         // header array, then body
}
async fn d(...) -> Result<Json<User>, StatusCode> { ... }    // Err becomes a status response

Errors

enum AppError { BadInput(String), NotFound }

impl IntoResponse for AppError {
    fn into_response(self) -> Response {
        let (status, message) = match self {
            AppError::BadInput(m) => (StatusCode::BAD_REQUEST, m),
            AppError::NotFound => (StatusCode::NOT_FOUND, "not found".to_string()),
        };
        (status, Json(json!({ "error": message }))).into_response()
    }
}

impl From<ParseIntError> for AppError { ... }   // lets ? convert

Extractor rejections are not your AppError. They return their own response before the handler runs.

State

let app = Router::new()
    .route("/notes", post(add_note))
    .with_state(AppState::default());     // Router<AppState> becomes Router<()>

async fn add_note(State(s): State<AppState>, body: String) -> StatusCode { ... }

impl FromRef<AppState> for Arc<Mutex<Vec<String>>> {  // substate
    fn from_ref(app: &AppState) -> Self { app.notes.clone() }
}

verified: axum::serve does not compile with a Router whose state is still unbound.

Middleware

CallScope
.layer(middleware::from_fn(f))Every route added before the call
get(h).route_layer(middleware::from_fn(f))Only that route, only when it matches
TraceLayer::new_for_http()Request and response logging (tower-http)
CorsLayer::permissive()Every origin. Lab only, not production
async fn guard(request: Request, next: Next) -> Result<Response, StatusCode> {
    if request.headers().contains_key("x-token") { Ok(next.run(request).await) }
    else { Err(StatusCode::UNAUTHORIZED) }
}

use axum::{body::Body, http::Request};

let response = app().oneshot(Request::builder().uri("/").body(Body::empty()).unwrap()).await.unwrap();
let bytes = response.into_body().collect().await.unwrap().to_bytes();

// Many requests on one service:
let mut service = app().into_service::<Body>();          // pin the body type, or E0283

Shipping

let listener = tokio::net::TcpListener::bind(addr).await?;
axum::serve(listener, app())
    .with_graceful_shutdown(async { tokio::signal::ctrl_c().await.ok(); })
    .await?;

Environment variables for settings. RUST_LOG through EnvFilter. Before production: bind address, CORS origins, body limits, timeouts. See lesson 10.
