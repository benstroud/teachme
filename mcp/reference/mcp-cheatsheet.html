Server: a tool

server.registerTool(
  "add_numbers",
  {
    description: "Add two numbers and return the sum.",
    inputSchema: z.object({ a: z.number(), b: z.number() }),
    outputSchema: z.object({ sum: z.number() }),
    annotations: { readOnlyHint: true },
  },
  async ({ a, b }) => ({
    content: [{ type: "text", text: String(a + b) }],
    structuredContent: { sum: a + b },
  }),
);

verified: a bad argument type returns isError: true with "Input validation error", not a protocol error.

Server: resource and prompt

server.registerResource("note", new ResourceTemplate("notes://{id}", { list: undefined }),
  { mimeType: "text/plain" },
  async (uri, { id }) => ({ contents: [{ uri: uri.href, mimeType: "text/plain", text: "..." }] }));

server.registerPrompt("summarize_note",
  { description: "...", argsSchema: z.object({ id: z.string() }) },
  async ({ id }) => ({ messages: [{ role: "user", content: { type: "text", text: "..." } }] }));

Serve

// stdio, one server per connection
serveStdio(() => buildServer(store));

// Streamable HTTP, localhost, with Origin and Host checks first
const handler = createMcpHandler(() => buildServer(store), { legacy: "reject" });
const rejected = originValidationResponse(request, localhostAllowedOrigins())
  ?? hostHeaderValidationResponse(request, localhostAllowedHostnames());
const response = rejected ?? (await handler.fetch(request));

verified: a foreign Origin returns 403 with JSON-RPC error -32000 "Invalid Origin".

Client

const client = new Client({ name: "app", version: "1.0.0" },
  { versionNegotiation: { mode: { pin: "2026-07-28" } } });  // default is "legacy"
await client.connect(new StdioClientTransport({ command: process.execPath, args: ["dist/server.js"] }));

const { tools } = await client.listTools();
const result = await client.callTool({ name: "add_numbers", arguments: { a: 2, b: 3 } });
if (result.isError) { /* show the text, do not retry blindly */ }
await client.readResource({ uri: "notes://n1" });
await client.getPrompt({ name: "summarize_note", arguments: { id: "n2" } });
